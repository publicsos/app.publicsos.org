```
-- Function to convert degrees to radians
CREATE OR REPLACE MACRO RADIANS(degrees) AS (degrees * PI() / 180);

-- Main query with Haversine distance calculation
WITH params AS (
  SELECT 
    -- Replace these values with your center point
    CAST(0.0 AS DOUBLE) as center_lat,
    CAST(YOUR_LONGITUDE AS DOUBLE) as center_lon,
    10.0 as radius_miles -- Search radius in miles
)
SELECT 
  type,
  "geometry/type",
  "geometry/coordinates/0" as longitude,
  "geometry/coordinates/1" as latitude,
  "properties/id",
  "properties/title",
  "properties/description",
  "properties/face",
  "properties/icon/url",
  "properties/icon/scaledSize/width",
  "properties/icon/scaledSize/height",
  "properties/icon/origin/x",
  "properties/icon/origin/y",
  "properties/icon/anchor/x",
  "properties/icon/anchor/y",
  -- Calculate distance in miles using Haversine formula
  3959 * 2 * ASIN(SQRT(
    POW(SIN(RADIANS(("geometry/coordinates/1" - p.center_lat) / 2)), 2) +
    COS(RADIANS(p.center_lat)) * 
    COS(RADIANS("geometry/coordinates/1")) * 
    POW(SIN(RADIANS(("geometry/coordinates/0" - p.center_lon) / 2)), 2)
  )) as distance_miles
FROM diagnoze.main.locations
CROSS JOIN params p
WHERE 3959 * 2 * ASIN(SQRT(
    POW(SIN(RADIANS(("geometry/coordinates/1" - p.center_lat) / 2)), 2) +
    COS(RADIANS(p.center_lat)) * 
    COS(RADIANS("geometry/coordinates/1")) * 
    POW(SIN(RADIANS(("geometry/coordinates/0" - p.center_lon) / 2)), 2)
  )) <= p.radius_miles
ORDER BY distance_miles;
```
